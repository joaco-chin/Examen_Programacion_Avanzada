ESTRUCTURA Arista:
    id, u, v, costo
    es_puente : booleano = falso

FUNCIÓN resolver_nuevas_autopistas(N, aristas[1..N+1], F[1..3]):
    // 1. Suma total y caso K = 0
    suma_total = suma de arista.costo para toda arista en aristas
    resuelto[1..3] = [falso, falso, falso]
    sol_k[1..3] = [-1, -1, -1]
    sol_eliminadas[1..3] = [lista_vacia, lista_vacia, lista_vacia]

    PARA i DE 1 HASTA 3:
        SI F[i] == suma_total ENTONCES
            resuelto[i] = verdadero
            sol_k[i] = 0
            sol_eliminadas[i] = []

    SI resuelto[1] Y resuelto[2] Y resuelto[3] ENTONCES
        RETORNAR (resuelto, sol_k, sol_eliminadas)

    // 2. Detectar puentes con Tarjan sobre el grafo original (N nodos, N+1 aristas)
    construir lista de adyacencia adj[1..N] con las aristas [1..N+1]
    tin[1..N] = arreglos inicializados en 0
    low[1..N] = arreglos inicializados en 0
    visitado[1..N] = arreglos inicializados en falso
    timer = 0

    // Nota: el grafo es conexo inicialmente, por lo que desde 1 se visita todo
    tarjan_puentes(nodo = 1, arista_padre_id = -1)

    // 3. Filtrar aristas NO puente y ordenar de mayor a menor costo
    candidatas = []
    PARA cada arista e EN aristas:
        SI NO e.es_puente ENTONCES
            agregar e a candidatas

    ordenar candidatas en orden DECRECIENTE por e.costo
    M_cand = tamaño(candidatas)

    // 4. Búsqueda con Branch and Bound en una única exploración
    eliminadas_actuales = []
    buscar(idx = 0, k = 0, costo_actual = suma_total)

    RETORNAR (resuelto, sol_k, sol_eliminadas)


FUNCIÓN tarjan_puentes(u, arista_padre_id):
    visitado[u] = verdadero
    timer = timer + 1
    tin[u] = low[u] = timer

    PARA cada (v, id_arista) EN adj[u]:
        SI id_arista == arista_padre_id ENTONCES
            CONTINUAR // Evita volver por la misma arista por la que llegamos

        SI visitado[v] ENTONCES
            // Arista de retroceso
            low[u] = min(low[u], tin[v])
        SINO
            tarjan_puentes(v, id_arista)
            low[u] = min(low[u], low[v])
            SI low[v] > tin[u] ENTONCES
                aristas[id_arista].es_puente = verdadero


FUNCIÓN buscar(idx, k, costo_actual):
    // Corte global si ya resolvimos los 3 objetivos
    SI resuelto[1] Y resuelto[2] Y resuelto[3] ENTONCES
        RETORNAR

    // Verificar si el costo actual coincide con algún objetivo pendiente (con k >= 1)
    SI k >= 1 ENTONCES
        PARA i DE 1 HASTA 3:
            SI NO resuelto[i] Y costo_actual == F[i] ENTONCES
                SI es_conexo_sin_eliminadas(N, aristas, eliminadas_actuales) ENTONCES
                    resuelto[i] = verdadero
                    sol_k[i] = k
                    sol_eliminadas[i] = copiar(eliminadas_actuales)

    // Si ya se resolvieron los 3 tras verificar, cortar
    SI resuelto[1] Y resuelto[2] Y resuelto[3] ENTONCES
        RETORNAR

    // Condiciones de borde: máximo K = 2 eliminaciones o fin de candidatas
    SI k == 2 O idx == M_cand ENTONCES
        RETORNAR

    // Calcular cotas de Branch and Bound en O(1)
    max_costo = costo_actual
    restantes_cupo = 2 - k
    min_costo = costo_actual

    SI restantes_cupo >= 1 Y idx < M_cand ENTONCES
        min_costo = min_costo - candidatas[idx].costo
    SI restantes_cupo == 2 Y (idx + 1) < M_cand ENTONCES
        min_costo = min_costo - candidatas[idx + 1].costo

    // Poda: ¿hay algún objetivo pendiente dentro de [min_costo, max_costo]?
    hay_objetivo_alcanzable = falso
    PARA i DE 1 HASTA 3:
        SI NO resuelto[i] Y (min_costo <= F[i]) Y (F[i] <= max_costo) ENTONCES
            hay_objetivo_alcanzable = verdadero
            ROMPER

    SI NO hay_objetivo_alcanzable ENTONCES
        RETORNAR // Podar toda la rama

    // Rama 1: QUITAR candidatas[idx]
    agregar candidatas[idx].id a eliminadas_actuales
    buscar(idx + 1, k + 1, costo_actual - candidatas[idx].costo)
    quitar último elemento de eliminadas_actuales

    // Rama 2: NO QUITAR candidatas[idx]
    buscar(idx + 1, k, costo_actual)


FUNCIÓN es_conexo_sin_eliminadas(N, aristas, eliminadas_ids):
    // Recorre las ciudades usando BFS/DFS ignorando las aristas en eliminadas_ids
    marcar todas las aristas como habilitadas, excepto las de eliminadas_ids
    visitados_bfs = BFS_o_DFS_desde(ciudad = 1, usando solo aristas habilitadas)
    RETORNAR (cantidad_visitados(visitados_bfs) == N)
